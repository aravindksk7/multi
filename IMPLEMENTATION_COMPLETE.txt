API USAGE EXAMPLE
--------------------------------------------------------------------------------

# PowerShell
$response = Invoke-RestMethod `
  -Uri "http://localhost:8000/api/xml-compare/jobs" `
  -Method Post `
  -Body (@{
    baseline_xml = "<testsuite><testcase name='TC1' status='pass'/></testsuite>"
    current_xml = "<testsuite><testcase name='TC1' status='fail'/></testsuite>"
    suite_name = "Regression"
    environment = "staging"
  } | ConvertTo-Json) `
  -ContentType "application/json"

# Returns:
{
  "id": 1,
  "status": "COMPLETED",
  "summary": {
    "added": 0,
    "removed": 0,
    "changed": 1,
    "total_differences": 1,
    "status": "FAILED"
  },
  "differences": [
    {
      "type": "CHANGED",
      "path": "/testsuite/testcase[@name='TC1']/@status",
      "baseline_value": "pass",
      "current_value": "fail"
    }
  ]
}
